// Enhanced E-Commerce Modal Functionality
(function() {
  const modal = document.getElementById('ecommerce-modal');
  const ecommerceCard = document.querySelector('[data-service="ecommerce"]');
  const closeBtn = modal.querySelector('.modal-close');
  const overlay = modal.querySelector('.modal-overlay');

  // Open modal
  if (ecommerceCard) {
    ecommerceCard.addEventListener('click', function(e) {
      e.stopPropagation();
      modal.classList.add('active');
      document.body.style.overflow = 'hidden';
    });
  }

  // Close modal function
  function closeModal() {
    modal.classList.remove('active');
    document.body.style.overflow = '';
  }

  // Close button
  if (closeBtn) {
    closeBtn.addEventListener('click', closeModal);
  }

  // Overlay click
  if (overlay) {
    overlay.addEventListener('click', closeModal);
  }

  // Escape key
  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape' && modal.classList.contains('active')) {
      closeModal();
    }
  });

  // Auto-open modal if URL has #ecommerce-modal hash
  if (window.location.hash === '#ecommerce-modal') {
    setTimeout(function() {
      modal.classList.add('active');
      document.body.style.overflow = 'hidden';
    }, 500); // Small delay to ensure page is loaded
  }
})();
