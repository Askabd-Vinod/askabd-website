  const serviceCards = document.querySelectorAll('.srv-card[data-service]');
  const modals = {
    'ecommerce': document.getElementById('ecommerce-modal'),
    'mobile': document.getElementById('mobile-modal'),
    'ai': document.getElementById('ai-modal'),
    'bi': document.getElementById('bi-modal'),
    'enterprise': document.getElementById('enterprise-modal'),
    'saas': document.getElementById('saas-modal'),
    'api': document.getElementById('api-modal'),
    'security': document.getElementById('security-modal'),
    'documentation': document.getElementById('documentation-modal')
  };

  // Open modal when clicking service card
  serviceCards.forEach(card => {
    card.addEventListener('click', function(e) {
      e.preventDefault();
      const serviceType = this.getAttribute('data-service');
      const modal = modals[serviceType];
      if (modal) {
        modal.style.display = 'flex';
        document.body.style.overflow = 'hidden';
      }
    });
  });

  // Close modal functionality
  Object.values(modals).forEach(modal => {
    if (modal) {
      const closeBtn = modal.querySelector('.modal-close');
      
      closeBtn.addEventListener('click', function() {
        modal.style.display = 'none';
        document.body.style.overflow = 'auto';
        // Remove hash from URL when closing modal
        if (window.location.hash) {
          history.replaceState(null, null, ' ');
        }
      });

      modal.addEventListener('click', function(e) {
        if (e.target === modal) {
          modal.style.display = 'none';
          document.body.style.overflow = 'auto';
          // Remove hash from URL when closing modal
          if (window.location.hash) {
            history.replaceState(null, null, ' ');
          }
        }
      });
    }
  });

  // Auto-open modal from URL hash
  const hash = window.location.hash.substring(1); // Remove the '#'
  if (hash && hash.endsWith('-modal')) {
    const serviceType = hash.replace('-modal', '');
    const modal = modals[serviceType];
    if (modal) {
      setTimeout(function() {
        modal.style.display = 'flex';
        document.body.style.overflow = 'hidden';
      }, 500);
    }
  }

  // ESC key to close modal
  document.addEventListener('keydown', function(e) {
    if (e.key === 'Escape') {
      Object.values(modals).forEach(modal => {
        if (modal && modal.style.display === 'flex') {
          modal.style.display = 'none';
          document.body.style.overflow = 'auto';
          // Remove hash from URL
          if (window.location.hash) {
            history.replaceState(null, null, ' ');
          }
        }
      });
    }
  });
});
</script>
